// Paste this block into Firebase console > kitzur-app1 > Storage > Rules,
// INSIDE the existing   match /b/{bucket}/o {   ...   }   block, next to the class-files rule.
// Don't delete anything that's already there.

    match /homework-audio/{assignmentId}/{fileName} {
      // A student can upload only their own recording, audio only, under 25 MB.
      allow write: if request.auth != null
        && (fileName == request.auth.token.email.lower() + '.webm'
            || fileName == request.auth.token.email.lower() + '.m4a'
            || fileName == request.auth.token.email.lower() + '.ogg')
        && request.resource.size < 25 * 1024 * 1024
        && request.resource.contentType.matches('audio/.*');

      // A student can play back their own recording. Teachers (anyone with a
      // document in the "teachers" collection) can play everyone's.
      allow read: if request.auth != null
        && (fileName == request.auth.token.email.lower() + '.webm'
            || fileName == request.auth.token.email.lower() + '.m4a'
            || fileName == request.auth.token.email.lower() + '.ogg'
            || firestore.exists(/databases/(default)/documents/teachers/$(request.auth.token.email)));
    }
